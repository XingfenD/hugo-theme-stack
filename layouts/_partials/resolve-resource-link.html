{{- /* Resolve a media reference to a hosted URL. Usage: {{ $url = partial "resolve-resource-link" (dict "value" $url) }} Resolution order: 1. Remote URLs (http/https) → returned unchanged 2. Other values → path.Join of the site base path (static files under baseURL subpaths) */ -}} {{- $value := .value -}} {{- if hasPrefix $value "http" -}} {{- $value -}} {{- else -}} {{- path.Join (urls.Parse site.BaseURL).Path (strings.TrimPrefix "/" $value) -}} {{- end -}}
